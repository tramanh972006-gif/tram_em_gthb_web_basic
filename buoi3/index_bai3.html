Phần 1

Biến (Variables)

Biến là nơi lưu trữ dữ liệu. Trong JavaScript, sử dụng let, const, hoặc var.

Let

Có thể thay đổi giá trị

Có phạm vi khối, không thể khai báo lại.

let name = "Trâm Anh";
name = "Tân"; // OK
let name = "Minh"; // Lỗi!

Const

Không thể thay đổi giá trị

Thường dùng cho giá trị không đổi.

const age = 20;
age = 21; // Lỗi!
const MAX = 100; // Hằng số

Var

Cách cũ (không nên dùng)

Có phạm vi global, dễ gây lỗi.

var x = 5;
x = 10; // OK
if(true) {
  var x = 20; // Ghi đè global
}

Ví dụ thực hành:

// Khai báo biến
let ho = "Đặng";
let ten = "Thị Trâm Anh";
const namSinh = 2000;

// Sử dụng biến
console.log(ho + " " + ten);
console.log("Năm sinh: " + namSinh);

// Tính tuổi
let tuoi = 2024 - namSinh;
console.log("Tuổi: " + tuoi);

Phần 2

Điều kiện (Conditionals)

Cho phép mã chạy hoặc không chạy dựa trên điều kiện thực hoặc sai.

If - Else

Chạy mã khi điều kiện đúng

let diem = 8;

if (diem >= 8) {
  console.log("Giỏi");
} else if (diem >= 6) {
  console.log("Khá");
} else {
  console.log("Chưa đạt");
}

Switch

Chọn một trường hợp trong nhiều trường hợp

let ngay = 3;

switch(ngay) {
  case 1:
    console.log("Thứ hai");
    break;
  case 2:
    console.log("Thứ ba");
    break;
  default:
    console.log("Ngày khác");
}

Toán tử Ternary

Điều kiện viết gọn một dòng

let tuoi = 20;

// Cách thường
let trangThai = tuoi >= 18 
  ? "Đã trưởng thành" 
  : "Chưa trưởng thành";

console.log(trangThai);

Toán tử so sánh

Các phép so sánh thường dùng

let x = 5;

console.log(x == "5");    // true
console.log(x === "5");   // false
console.log(x > 3);       // true
console.log(x <= 5);      // true
console.log(x != 10);     // true

Bài tập: Kiểm tra xem số có phải số chẵn không

let so = 7;

if (so % 2 === 0) {
  console.log(so + " là số chẵn");
} else {
  console.log(so + " là số lẻ");
}

// Hoặc dùng ternary
console.log(so % 2 === 0 ? "Chẵn" : "Lẻ");

Phần 3

Vòng lặp (Loops)

Lặp lại một đoạn mã nhiều lần cho đến khi điều kiện không còn đúng.

For Loop

Lặp từ đến với số lần xác định

// In ra 1 đến 5
for (let i = 1; i <= 5; i++) {
  console.log(i);
}

// In ra mảng
let fruits = ["Táo", "Cam"];
for (let j = 0; j < fruits.length; j++) {
  console.log(fruits[j]);
}

While Loop

Lặp khi điều kiện còn đúng

let i = 1;

while (i <= 5) {
  console.log(i);
  i++; // Tăng i lên 1
}

// Do-While: chạy ít nhất 1 lần
let x = 10;
do {
  console.log(x);
  x++;
} while (x <= 5);

ForEach

Duyệt qua từng phần tử mảng

let colors = ["Đỏ", "Xanh", "Vàng"];

// Cách 1: forEach
colors.forEach(function(mau) {
  console.log(mau);
});

// Cách 2: Arrow function
colors.forEach(mau => {
  console.log(mau);
});

For-In / For-Of

Duyệt key hoặc value

let obj = {ten: "Trâm", tuoi: 20};

// for-in: duyệt key
for (let key in obj) {
  console.log(key, obj[key]);
}

// for-of: duyệt value (mảng)
let arr = [10, 20, 30];
for (let val of arr) {
  console.log(val);
}

Bài tập: Tính tổng từ 1 đến 100

let tong = 0;

for (let i = 1; i <= 100; i++) {
  tong += i;
}

console.log("Tổng: " + tong); // 5050

// Hoặc dùng while
let i = 1, sum = 0;
while (i <= 100) {
  sum += i;
  i++;
}
console.log("Tổng: " + sum);

Phần 4

Hàm (Functions)

Hàm là một khối mã có tên, có thể tái sử dụng nhiều lần với các tham số khác nhau.

Hàm thường

Khai báo với function

function chao(ten) {
  return "Xin chào " + ten;
}

console.log(chao("Trâm"));
// "Xin chào Trâm"

Arrow Function

Cách viết gọn hơn

// Cách 1
const cong = (a, b) => {
  return a + b;
};

// Cách 2 (ngắn hơn)
const cong2 = (a, b) => a + b;

console.log(cong(5, 3)); // 8

Tham số mặc định

Giá trị default nếu không truyền

function tinhtien(gia, soluong = 1) {
  return gia * soluong;
}

console.log(tinhtien(100));      // 100
console.log(tinhtien(100, 5));   // 500

Hàm không return

Thực hiện hành động mà không trả về

function dophim(ten) {
  console.log("Đang phát: " + ten);
}

dophim("Avatar");
// Không trả về giá trị (undefined)

Bài tập: Hàm tính giai thừa

function tinhGiaiThua(n) {
  if (n <= 1) return 1;
  return n * tinhGiaiThua(n - 1);
}

console.log(tinhGiaiThua(5)); // 120

// Hoặc dùng vòng lặp
function giaiThua(n) {
  let kq = 1;
  for (let i = 2; i <= n; i++) {
    kq *= i;
  }
  return kq;
}

console.log(giaiThua(5)); // 120

Thực hành trên lớp

Bài tập JavaScript tổng hợp

Luyện tập function, array, object, loop, map/reduce/filter và format dữ liệu.

1. Tính điểm trung bình

function calculateAverage(scores) {
  let total = scores.reduce((sum, score) => sum + score, 0);
  return total / scores.length;
}

console.log(calculateAverage([8, 9, 7.5, 10]));

2. Kiểm tra số chẵn

function isEven(number) {
  return number % 2 === 0;
}

console.log(isEven(17));
console.log(isEven(20));

3. Tìm phần tử lớn nhất

function findMax(arr) {
  let max = arr[0];

  for (let i = 1; i < arr.length; i++) {
    if (arr[i] > max) {
      max = arr[i];
    }
  }

  return max;
}

console.log(findMax([12, 7, 23, 9, 25]));

4. Thao tác mảng sản phẩm

const products = [
  {id: 1, name: "Áo thun", price: 120000, stock: 18},
  {id: 2, name: "Giày thể thao", price: 560000, stock: 0},
  {id: 3, name: "Mũ lưỡi trai", price: 85000, stock: 5}
];

function formatProducts(products) {
  return products.map(product => {
    let status = product.stock > 0
      ? "Còn hàng"
      : "Hết hàng";

    return `${product.name} - ${product.price} đ - ${status}`;
  });
}

console.log(formatProducts(products));

for (let key in student) {
  console.log(key + ": " + student[key]);

function tinhTong(a, b) {
  return a + b;

function kiemTraChanLe(so) {
  if (so % 2 === 0) {

  } else {
    return "Lẻ";
  }

console.log(kiemTraChanLe(7)); // Lẻ
console.log(kiemTraChanLe(8)); // Chẵn

4. Tìm số lớn nhất trong mảng

let arr = [5, 12, 3, 8, 20];
let max = Math.max(...arr);
console.log(max); // 20

// Hoặc dùng vòng lặp
let max2 = arr[0];
for (let i = 1; i < arr.length; i++) {
  if (arr[i] > max2) {
    max2 = arr[i];
  }
}
console.log(max2); // 20

5. Tính tổng các phần tử trong mảy

let arr = [1, 2, 3, 4, 5];

// Cách 1: dùng reduce
let tong = arr.reduce((sum, x) => sum + x, 0);
console.log(tong); // 15

// Cách 2: dùng vòng lặp
let sum = 0;
for (let x of arr) {
  sum += x;
}
console.log(sum); // 15

6. Đếm số phần tử lớn hơn 10

let arr = [5, 12, 8, 15, 20, 3];

// Cách 1: dùng filter
let count = arr.filter(x => x > 10).length;
console.log(count); // 3

// Cách 2: dùng vòng lặp
let dem = 0;
for (let x of arr) {
  if (x > 10) dem++;
}
console.log(dem); // 3

7. Viết hàm đảo ngược chuỗi

function daoNguocChuoi(str) {
  return str.split("").reverse().join("");
}

console.log(daoNguocChuoi("Hello")); // olleH
console.log(daoNguocChuoi("Trâm")); // mârT

8. Lọc ra các số dương từ mảng hỗn hợp

let arr = [-5, 10, -3, 8, 0, 15];

// Dùng filter
let soDuong = arr.filter(x => x > 0);
console.log(soDuong); // [10, 8, 15]

9. Đếm số lần xuất hiện của một ký tự

function demKyTu(str, kyTu) {
  let dem = 0;
  for (let char of str) {
    if (char === kyTu) dem++;
  }
  return dem;
}

console.log(demKyTu("hello world", "l")); // 3
console.log(demKyTu("javascript", "a")); // 2

10. Quản lý danh sách sinh viên (Add, Edit, Delete)

let students = [
  {id: 1, name: "An", age: 20},
  {id: 2, name: "Bình", age: 21}
];

// Thêm sinh viên
function themSV(students, sv) {
  return [...students, sv];
}

// Xóa sinh viên
function xoaSV(students, id) {
  return students.filter(s => s.id !== id);
}

// Sửa sinh viên
function suaSV(students, id, updates) {
  return students.map(s => 
    s.id === id ? {...s, ...updates} : s
  );
}

console.log(themSV(students, {id: 3, name: "Trâm", age: 20}));
console.log(xoaSV(students, 1));
console.log(suaSV(students, 1, {age: 22}));

Math Object

Math trong JavaScript

Dùng để xử lý các phép toán nâng cao.

Math.random()

Sinh số ngẫu nhiên

console.log(Math.random());

// Random từ 1 -> 10
let number = Math.floor(Math.random() * 10) + 1;
console.log(number);

Math.floor()

Làm tròn xuống

console.log(Math.floor(4.9)); // 4
console.log(Math.floor(7.1)); // 7

Math.ceil()

Làm tròn lên

console.log(Math.ceil(4.1)); // 5
console.log(Math.ceil(7.01));

Math.round()

Làm tròn gần nhất

console.log(Math.round(4.4)); // 4
console.log(Math.round(4.5)); // 5

Math.max()

Tìm số lớn nhất

console.log(Math.max(4, 8, 2, 10));

Math.min()

Tìm số nhỏ nhất

console.log(Math.min(4, 8, 2, 10));

Break & Continue

Break và Continue

Điều khiển vòng lặp.

Break

Thoát khỏi vòng lặp

for (let i = 1; i <= 10; i++) {
  if (i === 5) {
    break;
  }

  console.log(i);
}

Continue

Bỏ qua lần lặp hiện tại

for (let i = 1; i <= 5; i++) {

  if (i === 3) {
    continue;
  }

  console.log(i);
}

So sánh

Các kiểu so sánh trong JavaScript

Phân biệt == và === rất quan trọng.

console.log(5 == "5");   
// true
// So sánh giá trị

console.log(5 === "5");  
// false
// So sánh giá trị + kiểu dữ liệu

console.log(10 != "10"); 
// false

console.log(10 !== "10");
// true
